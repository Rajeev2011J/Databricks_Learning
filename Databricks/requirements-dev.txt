# =============================================================================
# requirements-dev.txt
# Test and development dependencies for all Databricks Asset Bundles.
#
# THREE install scenarios — use the right one for your environment:
#
#   LOCAL-DEV (laptop):
#       pip install -r requirements-dev.txt
#       Installs everything including pyspark and delta-spark.
#
#   ADO-CI (Azure DevOps Linux agent):
#       Handled automatically by run_tests.sh --ci
#       Uses Nexus internal PyPI when NEXUSCLOUD_PYPI_URL is available.
#       Installs everything including pyspark and delta-spark.
#
#   DATABRICKS (cluster notebook):
#       Use requirements-cluster.txt instead (see below).
#       pyspark and delta-spark are provided by the cluster runtime —
#       installing them here would conflict with the cluster version.
#       The notebook runner (run_tests_notebook.py) handles this split.
#
# =============================================================================

# ---------------------------------------------------------------------------
# Core test runner
# ---------------------------------------------------------------------------
pytest==8.3.5
pytest-cov==7.1.0
pytest-mock==3.15.1

# ---------------------------------------------------------------------------
# Azure DevOps integration
# pytest-azurepipelines emits ##vso[] commands so results appear in the
# ADO "Tests" tab. It is a no-op outside of ADO — safe to install everywhere.
# ---------------------------------------------------------------------------
pytest-azurepipelines==1.0.5

# ---------------------------------------------------------------------------
# PySpark — local mode (LOCAL-DEV and ADO-CI only)
# NOT installed on Databricks clusters — the runtime provides these.
#
# Version compatibility matrix (must keep in sync):
#   pyspark 3.5.0  ↔  DBR 14.3 LTS  ↔  Python 3.11.x
#   pyspark 3.5.2  ↔  DBR 17.3 LTS  ↔  Python 3.11.x
#
# IMPORTANT — Python version constraint (from POC discovery):
#   Python 3.12+ breaks PySpark 3.5.x due to cloudpickle serialisation
#   changes. The ADO pipeline pins Python 3.11.x via UsePythonVersion@0.
#   Local developers must also use Python 3.11.x.
#   Check with: python --version   (must show 3.11.x)
#
# Pinned to 3.5.0 for widest cross-DBR compatibility.
# delta-spark 3.2.1 is the compatible version for Spark 3.5.x.
# ---------------------------------------------------------------------------
pyspark==3.5.0
delta-spark==3.2.1

# ---------------------------------------------------------------------------
# pytest-html — human-readable test execution report
# Produces a self-contained HTML file showing pass/fail/skip per test with
# durations, tracebacks, and system environment details.
# Used by the pipeline to publish test-report.html as a build artefact.
# ---------------------------------------------------------------------------
pytest-html==3.2.0
pytest-metadata==3.0.0

# ---------------------------------------------------------------------------
# DataFrame assertion library
# ---------------------------------------------------------------------------
chispa==0.12.0

# ---------------------------------------------------------------------------
# Databricks SDK
# Needed so `from databricks.sdk.runtime import ...` resolves at import time
# (we stub the runtime objects, but the package namespace must exist).
# Already present on Databricks clusters — safe to install; won't conflict.
# ---------------------------------------------------------------------------
databricks-sdk==0.32.0

# ---------------------------------------------------------------------------
# YAML parsing — used by YAML-validation tests (clusters, PartyCatalog)
# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------

setuptools==75.8.2

# ---------------------------------------------------------------------------
# Optional plugins

pytest-sugar==1.0.0        # progress bar
pytest-xdist==3.6.1        # parallel: pytest -n auto  (unit tests only)
pytest-timeout==2.4.0      # @pytest.mark.timeout(30)

# ---------------------------------------------------------------------------
# Linting (not required to run tests)
# ---------------------------------------------------------------------------
ruff==0.6.9
